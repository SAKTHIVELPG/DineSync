ESP32 Arduino C++ Firmware (dinesync_esp32.ino)

Ready-to-flash C++ sketch for physical ESP32 boards with HC-SR04 ultrasonic sensors

// DINESYNC ESP32 Table Node Firmware
#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>

const char* WIFI_SSID     = "YOUR_RESTAURANT_IOT_SSID";
const char* WIFI_PASSWORD = "YOUR_WIFI_PASSWORD";
const char* SERVER_URL    = "http://192.168.1.100:8000/api/v1/sensors/telemetry";

const char* SENSOR_ID     = "ESP32-NODE-04";
const int   TABLE_NUMBER  = 4;
const int   PIN_TRIG      = 5;
const int   PIN_ECHO      = 18;

void setup() {
  Serial.begin(115200);
  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    digitalWrite(PIN_TRIG, HIGH);
    delayMicroseconds(10);
    digitalWrite(PIN_TRIG, LOW);
    long duration = pulseIn(PIN_ECHO, HIGH);
    float distanceCm = (duration * 0.0343) / 2.0;
    bool occupied = (distanceCm < 45.0);

    HTTPClient http;
    http.begin(SERVER_URL);
    http.addHeader("Content-Type", "application/json");
    
    StaticJsonDocument<200> doc;
    doc["sensor_id"] = SENSOR_ID;
    doc["table_id"] = TABLE_NUMBER;
    doc["distance_cm"] = distanceCm;
    doc["occupied"] = occupied;
    doc["battery_level"] = 95;
    doc["signal_rssi"] = WiFi.RSSI();
    
    String json;
    serializeJson(doc, json);
    http.POST(json);
    http.end();
  }
  delay(3000);
}
